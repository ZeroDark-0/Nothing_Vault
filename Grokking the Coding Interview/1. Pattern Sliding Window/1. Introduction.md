--> Sliding Window 

Question -> `Array: [1, 3, 2, 6, -1, 4, 1, 8, 2], K=5` an array is given with the value of k we have to find average of all the contiguous of the subarray of size k.

To do this there are two approach first using 2 loops meaning have O(NxK) complexity and there other one is sliding window with O(N)..

With the O(NxK) complexity


```cpp
#include <iostream>
using namespace std;

int main(){
    int arr[] = {1, 3, 2, 6, -1, 4, 1, 8, 2};
    int k = 5;
    int ans[ sizeof(arr)/sizeof(arr[0])-k+1];
    for(int i = 0;i < sizeof(arr)/sizeof(arr[0])-k+1;i++){
        int sum = 0;
        for(int j = i;j<i+k;j++){
            sum+=arr[j];
        }
        sum/=k;
        ans[i] = sum;
    }
    for (int ele : arr) { cout << ele << " "; } cout << endl;
}
```

Now how to do this with only O(N) times complexity

```cpp

#include <iostream>
#include <vector>

using namespace std;

int main() {
    vector<int> nums = {1, 3, 2, 6, -1, 4, 1, 8, 2};
    int k = 5;
    vector<double> result;
    double windowSum = 0;
    int windowStart = 0;

    for (int windowEnd = 0; windowEnd < nums.size(); windowEnd++) {
        windowSum += nums[windowEnd];  

        if (windowEnd >= k - 1) {
            result.push_back(windowSum / k);      
            windowSum -= nums[windowStart];       
            windowStart++;                        
        }
    }

    cout << "Averages of subarrays of size " << k << ": ";
    for (double avg : result) {
        cout << avg << " ";
    }
    cout << endl;

    return 0;
}
```
